**Goal:** Complete Phase 2 of the Shiori cognitive platform (MVP Complete / Demo-Ready) offline-first across all four roles: merge Patient UI games and persist to Room DB, add Daily Routine checklist, implement Tier 1 Staircase Adaptive Difficulty, implement Multilingual Audio Prompt Manager (EN, HI, KN), and implement the Caregiver PIN gate and local analytics dashboard with DPDP notice.

**Architecture:** MVI/MVVM pattern with Jetpack Compose UI, Room local encrypted-ready SQLite database, deterministic staircase difficulty scaling on device, native Android TextToSpeech, and PIN-gated caregiver mode.

**Tech Stack:** Kotlin 1.9/2.0, Jetpack Compose, Material3, Room DB, Hilt DI, Android TextToSpeech, JUnit 4, Kotlin Coroutines & Flow.

## Global Constraints

- Android API: `minSdk = 26`, `targetSdk = 34`, `compileSdk = 34`.
- Offline First: Zero required network calls for all core patient and caregiver features.
- High Contrast / Accessibility: Elderly-friendly fonts (≥18sp body, ≥24sp headers), touch targets ≥64dp.
- Data Security: All session and reminder data stored locally in Room DB with `SyncStatus.LOCAL_ONLY`.
- Architecture Layers: Clean separation between `domain/`, `data/`, `ml/`, `voice/`, and `ui/`.

```kotlin
package com.shiori.app.ml

import com.shiori.app.domain.model.GameSession
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class Tier1StaircaseDifficultyAdapter @Inject constructor() : DifficultyAdapterModel {

    override fun calculateNextDifficulty(recentSessions: List<GameSession>): Int {
        if (recentSessions.isEmpty()) {
            return 1
        }

        val lastSession = recentSessions.last()
        val currentLevel = lastSession.difficultyLevel.coerceIn(1, 5)

        // Rule 1: Single failure drops level (-1)
        if (lastSession.accuracy < 0.50f) {
            return (currentLevel - 1).coerceAtLeast(1)
        }

        // Rule 2: Two consecutive high scores step up (+1)
        if (recentSessions.size >= 2) {
            val secondLast = recentSessions[recentSessions.size - 2]
            if (lastSession.accuracy >= 0.80f && secondLast.accuracy >= 0.80f) {
                return (currentLevel + 1).coerceAtMost(5)
            }
        }

        return currentLevel
    }
}
```

```kotlin
package com.shiori.app.ui.patient

import org.junit.Assert.assertEquals
import org.junit.Test

class DailyRoutineModelTest {
    data class RoutineItem(val id: String, val title: String, val isCompleted: Boolean)

    private fun calculateCompletion(items: List<RoutineItem>): Float {
        if (items.isEmpty()) return 0f
        val completed = items.count { it.isCompleted }
        return completed.toFloat() / items.size
    }

    @Test
    fun testCompletionPercentage() {
        val items = listOf(
            RoutineItem("1", "Walk", true),
            RoutineItem("2", "Water", true),
            RoutineItem("3", "Meds", false),
            RoutineItem("4", "Sleep", false)
        )
        assertEquals(0.5f, calculateCompletion(items), 0.01f)
    }
}
```
